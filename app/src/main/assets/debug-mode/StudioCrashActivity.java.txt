package com.studiomobile.debugcatcher;

import android.app.Activity;
import android.content.ClipData;
import android.content.ClipboardManager;
import android.content.Context;
import android.graphics.Color;
import android.graphics.Typeface;
import android.graphics.drawable.GradientDrawable;
import android.os.Bundle;
import android.view.Gravity;
import android.view.View;
import android.view.ViewGroup;
import android.widget.Button;
import android.widget.LinearLayout;
import android.widget.ScrollView;
import android.widget.TextView;
import android.widget.Toast;

import java.io.ByteArrayOutputStream;
import java.io.File;
import java.io.FileInputStream;

/**
 * Full-screen white crash page shown by Studio Mobile Debug Mode. Runs in its own process so it
 * stays alive after the crashed app process is killed. Built in code: it needs no resources.
 */
public final class StudioCrashActivity extends Activity {

    @Override protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        final String log = loadLog();
        final float density = getResources().getDisplayMetrics().density;
        int pad = (int) (16 * density);

        LinearLayout root = new LinearLayout(this);
        root.setOrientation(LinearLayout.VERTICAL);
        root.setBackgroundColor(Color.WHITE);
        root.setPadding(pad, pad * 2, pad, pad);

        TextView title = new TextView(this);
        final boolean frozen = log.startsWith("=== APP FROZE");
        final boolean anr = log.startsWith("=== APP WAS NOT RESPONDING");
        title.setText(frozen ? "The app is not responding" : anr ? "The app was not responding" : "The app crashed");
        title.setTextColor(Color.parseColor("#B00020"));
        title.setTextSize(22);
        title.setTypeface(Typeface.DEFAULT_BOLD);
        root.addView(title);

        TextView hint = new TextView(this);
        hint.setText(frozen
            ? "The screen of the app stopped answering. The log shows where it is stuck. The app may still be running behind this page: Close returns to it. Copy the log and send it to your AI assistant."
            : anr
            ? "Last time, Android ended the app because it stopped responding. This is a record of that earlier event. Copy the log and send it to your AI assistant."
            : "Debug Mode caught this before it could close silently. Copy the log and send it to your AI assistant.");
        hint.setTextColor(Color.parseColor("#555555"));
        hint.setTextSize(14);
        hint.setPadding(0, pad / 2, 0, pad / 2);
        root.addView(hint);

        LinearLayout buttons = new LinearLayout(this);
        buttons.setOrientation(LinearLayout.HORIZONTAL);
        buttons.setGravity(Gravity.CENTER_VERTICAL);
        Button copy = new Button(this);
        copy.setText("Copy log");
        copy.setAllCaps(false);
        copy.setOnClickListener(new View.OnClickListener() {
            @Override public void onClick(View v) {
                ClipboardManager clipboard = (ClipboardManager) getSystemService(Context.CLIPBOARD_SERVICE);
                if (clipboard != null) clipboard.setPrimaryClip(ClipData.newPlainText("Crash log", log));
                Toast.makeText(StudioCrashActivity.this, "Log copied", Toast.LENGTH_SHORT).show();
            }
        });
        Button close = new Button(this);
        close.setText("Close");
        close.setAllCaps(false);
        close.setOnClickListener(new View.OnClickListener() {
            @Override public void onClick(View v) { finishAndRemoveTask(); android.os.Process.killProcess(android.os.Process.myPid()); }
        });
        buttons.addView(copy, new LinearLayout.LayoutParams(0, ViewGroup.LayoutParams.WRAP_CONTENT, 1f));
        buttons.addView(close, new LinearLayout.LayoutParams(0, ViewGroup.LayoutParams.WRAP_CONTENT, 1f));
        root.addView(buttons);

        ScrollView scroll = new ScrollView(this);
        GradientDrawable box = new GradientDrawable();
        box.setColor(Color.parseColor("#F5F5F5"));
        box.setCornerRadius(8 * density);
        scroll.setBackground(box);
        TextView body = new TextView(this);
        body.setText(log);
        body.setTextColor(Color.parseColor("#111111"));
        body.setTextSize(12);
        body.setTypeface(Typeface.MONOSPACE);
        body.setTextIsSelectable(true);
        body.setPadding(pad / 2, pad / 2, pad / 2, pad / 2);
        scroll.addView(body);
        LinearLayout.LayoutParams scrollParams = new LinearLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, 0, 1f);
        scrollParams.topMargin = pad / 2;
        root.addView(scroll, scrollParams);

        setContentView(root);
    }

    private String loadLog() {
        String fromIntent = getIntent() == null ? null : getIntent().getStringExtra(StudioDebugProvider.EXTRA_LOG);
        if (fromIntent != null && fromIntent.length() > 0) return fromIntent;
        try {
            File file = new File(getFilesDir(), StudioDebugProvider.LOG_FILE_NAME);
            FileInputStream in = new FileInputStream(file);
            try {
                ByteArrayOutputStream out = new ByteArrayOutputStream();
                byte[] buffer = new byte[8192]; int count;
                while ((count = in.read(buffer)) != -1) out.write(buffer, 0, count);
                return out.toString("UTF-8");
            } finally { in.close(); }
        } catch (Exception error) {
            return "No log found: " + error;
        }
    }

    @Override public void onBackPressed() { finishAndRemoveTask(); android.os.Process.killProcess(android.os.Process.myPid()); }
}
